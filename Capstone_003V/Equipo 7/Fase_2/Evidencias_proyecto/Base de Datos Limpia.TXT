-- ================================================
-- Tabla: usuarios
-- ================================================
CREATE TABLE usuarios (
  id_usuario SERIAL PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL,
  email VARCHAR(100) UNIQUE NOT NULL,
  password_hash TEXT NOT NULL,
  rol VARCHAR(50) NOT NULL CHECK (rol IN ('admin', 'cocinero', 'auditor')),
  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ================================================
-- Tabla: proveedores
-- ================================================
CREATE TABLE proveedores (
  id_proveedor SERIAL PRIMARY KEY,
  nombre VARCHAR(150) NOT NULL,
  contacto VARCHAR(150) NOT NULL,
  telefono VARCHAR(50) NOT NULL,
  email VARCHAR(100) NOT NULL UNIQUE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT chk_telefono CHECK (telefono ~ '^\+?[0-9]+$'),
  CONSTRAINT chk_email CHECK (email ~ '^[^@]+@[^@]+\.[^@]+$')
);

-- ================================================
-- Tabla: productos
-- ================================================
CREATE TABLE productos (
  id_producto SERIAL PRIMARY KEY,
  nombre VARCHAR(150) NOT NULL,
  codigo VARCHAR(50) UNIQUE NOT NULL,
  categoria VARCHAR(100) NOT NULL,
  unidad_medida VARCHAR(50) NOT NULL,
  id_proveedor INT NOT NULL REFERENCES proveedores(id_proveedor) ON DELETE SET NULL,
  stock_actual INT DEFAULT 0,
  stock_minimo INT DEFAULT 0,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ================================================
-- Tabla: inventario
-- ================================================
CREATE TABLE inventario (
  id_inventario SERIAL PRIMARY KEY,
  id_producto INT NOT NULL REFERENCES productos(id_producto),
  stock_actual NUMERIC NOT NULL DEFAULT 0,
  fecha_actualizacion TIMESTAMP DEFAULT NOW(),
  fecha_ingreso TIMESTAMP DEFAULT NOW(),
  fecha_caducidad DATE
);

-- ================================================
-- Tabla: movimientos_insumo
-- ================================================
CREATE TABLE movimientos_insumo (
  id_movimiento SERIAL PRIMARY KEY,
  id_producto INT NOT NULL REFERENCES productos(id_producto),
  cantidad NUMERIC NOT NULL,
  tipo VARCHAR(20) NOT NULL CHECK (tipo IN ('entrada','salida','ajuste')),
  fecha TIMESTAMP DEFAULT NOW(),
  id_usuario INT NOT NULL REFERENCES usuarios(id_usuario)
);

-- ================================================
-- Tabla: checklists
-- ================================================
CREATE TABLE checklists (
  id_checklist SERIAL PRIMARY KEY,
  fecha DATE NOT NULL DEFAULT CURRENT_DATE,
  turno VARCHAR(20) NOT NULL CHECK (turno IN ('Mañana','Tarde','Noche')),
  id_usuario INT NOT NULL REFERENCES usuarios(id_usuario) ON DELETE CASCADE,
  categoria VARCHAR(100) NOT NULL DEFAULT 'Cocina',
  estado VARCHAR(20) NOT NULL CHECK (estado IN ('pendiente','completo','incompleto')) DEFAULT 'pendiente',
  observaciones TEXT,
  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ================================================
-- Tabla: checklist_items
-- ================================================
CREATE TABLE checklist_items (
  id_item SERIAL PRIMARY KEY,
  id_checklist INT NOT NULL REFERENCES checklists(id_checklist) ON DELETE CASCADE,
  descripcion VARCHAR(200) NOT NULL,
  resultado VARCHAR(20) NOT NULL CHECK (resultado IN ('ok','no_ok','na')),
  observacion TEXT,
  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  actualizado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ================================================
-- Tabla: auditoria
-- ================================================
CREATE TABLE auditoria (
  id_auditoria SERIAL PRIMARY KEY,
  tabla_afectada VARCHAR(50) NOT NULL,
  id_registro INT NOT NULL,
  accion VARCHAR(20) NOT NULL CHECK (accion IN ('INSERT','UPDATE','DELETE')),
  id_usuario INT NOT NULL REFERENCES usuarios(id_usuario),
  detalles TEXT,
  fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ================================================
-- Tabla: checklist_template_items
-- ================================================
CREATE TABLE checklist_template_items (
  id_template_item SERIAL PRIMARY KEY,
  descripcion TEXT NOT NULL
);

-- ================================================
-- Tabla: alertas
-- ================================================
CREATE TABLE alertas (
  id_alerta SERIAL PRIMARY KEY,
  tipo VARCHAR(50) NOT NULL CHECK (tipo IN ('stock_bajo','temperatura','checklist_incompleto')),
  descripcion TEXT NOT NULL,
  fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  estado VARCHAR(20) NOT NULL DEFAULT 'pendiente' CHECK (estado IN ('pendiente','resuelta','ignoradas')),
  id_referencia INT,
  id_usuario INT REFERENCES usuarios(id_usuario) ON DELETE SET NULL
);

CREATE INDEX idx_alertas_tipo_estado ON alertas(tipo, estado);
CREATE INDEX idx_alertas_fecha ON alertas(fecha);

-- ================================================
-- Tabla: produccion_alimentos
-- ================================================
CREATE TABLE produccion_alimentos (
  id_produccion SERIAL PRIMARY KEY,
  fecha DATE NOT NULL,
  id_usuario INT REFERENCES usuarios(id_usuario),
  responsable VARCHAR(100),
  turno VARCHAR(20) CHECK (turno IN ('Mañana','Tarde','Noche')),
  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- ================================================
-- Tabla: produccion_detalle
-- ================================================
CREATE TABLE produccion_detalle (
  id_detalle SERIAL PRIMARY KEY,
  id_produccion INT REFERENCES produccion_alimentos(id_produccion) ON DELETE CASCADE,
  id_producto INT REFERENCES productos(id_producto),
  cantidad NUMERIC NOT NULL,
  temp_coccion NUMERIC,
  hora TIME,
  observaciones TEXT
);
